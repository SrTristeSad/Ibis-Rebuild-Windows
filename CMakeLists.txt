cmake_minimum_required(VERSION 3.20)
project(Android1412DesktopRebuild VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This stage targets Windows x64.")
endif()
add_library(rebuild_core SHARED native/core.cpp)
target_compile_features(rebuild_core PRIVATE cxx_std_17)
set_target_properties(rebuild_core PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(MSVC)
  target_compile_options(rebuild_core PRIVATE /O2 /GS- /GR- /Zl /fp:precise)
  target_link_options(rebuild_core PRIVATE /NOENTRY /NODEFAULTLIB)
else()
  target_compile_options(rebuild_core PRIVATE -O2 -ffreestanding -fno-fast-math -fno-builtin -fno-exceptions -fno-rtti -fno-stack-protector)
  target_link_options(rebuild_core PRIVATE -nostdlib -Wl,/noentry,/nodefaultlib)
endif()
find_package(Python3 COMPONENTS Interpreter)
include(CTest)
if(BUILD_TESTING AND Python3_FOUND)
  add_test(NAME native_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_core.py)
  set_tests_properties(native_contract PROPERTIES TIMEOUT 35 ENVIRONMENT "RB_CORE_DLL=$<TARGET_FILE:rebuild_core>")
endif()
